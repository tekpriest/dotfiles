** File Operations (genghis port)
File manipulation commands matching =<leader>m*= from nvim-genghis.
#+begin_src elisp
(defun tp/copy-relative-path ()
  (interactive)
  (if (buffer-file-name)
      (let ((path (file-relative-name (buffer-file-name) (or (projectile-project-root) default-directory))))
        (kill-new path)
        (message "Copied: %s" path))
    (user-error "Buffer has no file")))

(defun tp/copy-filename ()
  (interactive)
  (if (buffer-file-name)
      (let ((name (file-name-nondirectory (buffer-file-name))))
        (kill-new name)
        (message "Copied: %s" name))
    (user-error "Buffer has no file")))

(defun tp/rename-file ()
  (interactive)
  (unless (buffer-file-name) (user-error "Buffer has no file"))
  (let* ((old (buffer-file-name))
         (new (read-file-name "Rename to: " (file-name-directory old) nil nil
                              (file-name-nondirectory old))))
    (rename-file old new 1)
    (set-visited-file-name new t t)))

(defun tp/move-file ()
  (interactive)
  (unless (buffer-file-name) (user-error "Buffer has no file"))
  (let* ((old (buffer-file-name))
         (dir (read-directory-name "Move to: "))
         (new (expand-file-name (file-name-nondirectory old) dir)))
    (rename-file old new 1)
    (set-visited-file-name new t t)))

(defun tp/duplicate-file ()
  (interactive)
  (unless (buffer-file-name) (user-error "Buffer has no file"))
  (let* ((old (buffer-file-name))
         (new (read-file-name "Duplicate to: " (file-name-directory old) nil nil
                              (concat "copy-of-" (file-name-nondirectory old)))))
    (copy-file old new 1)
    (find-file new)))

(defun tp/trash-file ()
  (interactive)
  (unless (buffer-file-name) (user-error "Buffer has no file"))
  (let ((file (buffer-file-name)))
    (when (yes-or-no-p (format "Trash %s? " (file-name-nondirectory file)))
      (move-file-to-trash file)
      (kill-buffer))))

(defun tp/chmodx ()
  (interactive)
  (unless (buffer-file-name) (user-error "Buffer has no file"))
  (let ((file (buffer-file-name)))
    (set-file-modes file (logior (file-modes file) #o111))
    (message "chmod +x %s" (file-name-nondirectory file))))

(defun tp/create-new-file ()
  (interactive)
  (let ((dir (if (buffer-file-name)
                 (file-name-directory (buffer-file-name))
               default-directory)))
    (find-file (read-file-name "New file: " dir))))

(defun tp/move-selection-to-new-file (beg end)
  (interactive "r")
  (let* ((text (buffer-substring-no-properties beg end))
         (dir  (if (buffer-file-name) (file-name-directory (buffer-file-name)) default-directory))
         (dest (read-file-name "Move selection to: " dir)))
    (delete-region beg end)
    (with-temp-file dest (insert text))
    (message "Moved selection → %s" dest)))

(map! :leader
      :n "mp" #'tp/copy-relative-path
      :n "mc" #'tp/copy-filename
      :n "mr" #'tp/rename-file
      :n "mm" #'tp/move-file
      :n "mx" #'tp/chmodx
      :n "md" #'tp/duplicate-file
      :n "me" #'tp/trash-file
      :n "mn" #'tp/create-new-file
      :v "mm" #'tp/move-selection-to-new-file)
